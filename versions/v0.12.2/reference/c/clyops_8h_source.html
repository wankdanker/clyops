1/*
2 * clyops — declarative CLI parsing for C11 (POSIX). Behavior follows spec/SPEC.md.
3 *
4 * clyops_t* cli = clyops_new(NULL);
5 * clyops_arg(cli, "input", .description = "Input file", .rule = "path");
6 * clyops_opt(cli, "PORT", "port", "8080", .short_name = 'p', .description = "Server port", .group = "Network", .rule = "port");
7 * clyops_opt(cli, "VERBOSE", "verbose", "flag", .short_name = 'v', .description = "Verbose output");
8 * clyops_run(cli, argc, argv);
9 * printf("%s %lld %d\n", clyops_get(cli, "input"), clyops_get_int(cli, "PORT"), clyops_get_bool(cli, "VERBOSE"));
10 * clyops_free(cli);
11 *
12 * The registration macros wrap their trailing designated initializers in a
13 * clyops_meta_t compound literal, so every option is one line with named
14 * optional fields. The plain clyops_add_* functions accept a reusable
15 * clyops_meta_t as well.
16 */
